      rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ランキング（Rhythm Drums）
    match /rhythm_scores/{scoreId} {
      allow read: if true;
      allow create: if
        request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 10
        && request.resource.data.score is number
        && request.resource.data.score >= 0;
      allow update, delete: if false;
    }

    // 魚種別ランキング（延べ竿日和）
    match /nobezao_scores/{scoreId} {
      allow read: if true;
      allow create: if
        request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 10
        && request.resource.data.speciesId is string
        && request.resource.data.speciesId.size() >= 1
        && request.resource.data.size is number
        && request.resource.data.size > 0
        && request.resource.data.size <= 1000
        && request.resource.data.weight is number
        && request.resource.data.weight > 0
        && request.resource.data.weight <= 10000;
      allow update, delete: if false;
    }

    // 閲覧カウンター（ORBITなど）
    match /page_views/{pageId} {
      allow read: if true;
      allow create: if request.resource.data.count == 1;
      allow update: if request.resource.data.count == resource.data.count + 1;
      allow delete: if false;
    }

    // それ以外のコレクションは全て禁止
    match /{document=**} {
      allow read, write: if false;
    }
  }

        遊び方と、本式の作法

        1. 仕掛けを入れて待つ。打ち込む場所は水面をタップして決める。タップしたその場所へ仕掛けが飛び、ウキが立つ。空や山、岸の草など水面以外をタップしても何も起きない（「仕掛けを入れる」ボタンは、いまの場所――初めてなら水面の中央寄り――へ打ち込む）。アタリは最速で10秒、遅くても1分半（90秒）以内に必ず出る（1分半を超えてアタリなしにはならない）。同じ場所に打ち続けるのが寄せの作法で、打ち続けるほど魚が寄ってきて、10秒〜1分半の中でも早めにアタリが出やすくなる（エサや季節・時間帯を変えると寄せは解ける）。鯉が近づくと泡付け（小さな泡）やウキの不自然な揺れが出る。風が渡るたび、水面にはさざ波と明るい風の通り道が走り、ウキも波に揺れてわずかに流される。風が強いときはウキが風下へはっきり傾くので、傾き具合で風の強さが目で分かる（凪ではほぼ真っ直ぐに立つ）。
        2. 餌が付いているかは、ウキの目盛りの色で見る。餌が付いている間は餌の重さでウキが深く沈み、緑の目盛りがちょうど水面に出る。餌が溶け・小さくなるにつれてウキは少しずつ浮き上がり、餌が無くなると赤い目盛りが水面から出る――それが「餌なし」の合図で、そのままではアタリは来ない。餌にはそれぞれ「餌持ち」があり、パンは約45秒で取れやすく、練り餌は約1分半で溶け、ミミズは約2分半、コーンは約4分と固く長持ちする（重いコーンは深めに、軽いパンは浅めに沈む）。「餌付け」を押すと、仕掛けを手繰り寄せて新しい餌を付けるが、仕掛けはすぐには投入されない。餌を付け終えた仕掛けは手元に置かれたままになり、水面をタップした場所へ初めて打ち込まれる（寄せは保たれる）。設定でエサを変えたときも同じで、仕掛けを回収してウキを一度消し、新しい餌を付けたら、水面タップで打ち直す。新しい餌が付けばウキはまた深く沈む。餌が切れる前でも、ウキが浮いてきたら早めに付け替えて打ち直すのが本式の作法。
        3. 前アタリでは合わせないのが本式。鯉は底のエサを泥ごと吸い込んで吐き出しながら食べるので、まずウキが数cm沈む「前アタリ」が来る。ここで合わせると大抵は空振りになるが、魚がエサを深く吸い込んでいたときは時々そのまま掛かることもある（前アタリの後半ほど掛かりやすい）。本式は、じっと待つこと。ただし確率で（およそ3回に1回）前アタリもなく、いきなり本アタリが来ることもある。気を抜いていると、消し込みの一瞬を逃す。
        4. 本アタリで合わせる――合わせは「押して、離す」。打ち込んだあとに出るボタンは「竿を持つ」ボタン。あらかじめボタンを押し続けて竿を持っておき、本アタリの瞬間に指を離すと、その瞬間に合わせが決まる。押しただけでは合わせにならない（押し続けている間は竿を持っているだけ）。ウキが完全に消し込む、または食い上げて浮き上がったら、持っていた指を離すこと。竿を持つのが遅れて本アタリの最中に押し始めたときは、押してすぐ離せばその場で合わせられる。特に消し込みでウキが水面から完全に消えたままの間は、魚がまだエサをくわえているので、少し遅れて離しても掛かる。前アタリの間は離さず、竿を持ったまま本アタリを待つのが本式。竿を持っていない時に強いアタリ（本アタリ）が来て、そのまま持たずにいると、竿が沖へ飛ばされて仕掛けごと失ってしまう。そうなったら餌付けからやり直しになるので、待ちの間は竿を持ち続けること（お邪魔の間は合わせ自体ができないので、その間だけは飛ばされない）。
        5. やり取りは竿を立てて溜める。リールがないので糸は出せない。「竿を立てる」を長押しして張りを保ち、緑の範囲を外さないように。緑の範囲に保てている間は魚がぐっと近づいてきて、走られても少しずつ寄せられる。張りすぎればハリスが切れ、緩めすぎればバレる。張りのシビアさは魚ごとに違う：タナゴやオイカワは適正範囲が広くゆるめだが、マゴイは狭く、大型（50cm超・80cm超）になるほど範囲がさらに狭まり、張りの上下も速く、限界の猶予も短くなる。掛かった瞬間に「適正 ○〜○％」とシビアさが表示されるので、まずそれを見てから小刻みに調整する。魚が疲れて近づいたらタモ入れ。タモは頭から。
        6. 操作はボタンと水面タップ。打ち込みは水面をタップした場所へ。打ち直したあとも、水面をタップするまで仕掛けは投入されない。合わせは「竿を持つ」方式：ボタンを押し続けて竿を持ち、本アタリの瞬間に指を離すと合わせる（水面も同じ作法で、触れている間は竿を持ち、離すと合わせる。水面以外は反応しない。スペースキーも、押している間は竿を持ち、離すと合わせる）。やり取り中の糸の張りは「竿を立てる（長押し）」ボタンで調整し、押している間は張り、離すと緩む。お邪魔中は「竿を持つ」ボタンも「餌付け」も使えない（水面・スペースキーでの合わせも同じ）。まずボールを全部消してから合わせること。
        7. お邪魔モード（常時ON）。釣りを始めてから、アタリとまったく同じ確率（最速10秒〜最長1分半に1回）で、おじいさん・おばあさんの話しかけ、悪ガキのルアー投げ込み、野良猫、写真好きの観光客、釣りの先輩、動画配信者、押しの強いセールスマンなどが現れる。静かさに負けて居眠りしてしまうこともある。お邪魔のセリフは、キャラごとに違う声で実際に読み上げられる（おじいさんは低くゆっくり、おばあさんは少し高く、悪ガキは高く速く、居眠りは眠そうに遅く、など）。画面にボールが10〜30個降ってくる（厄介な相手ほど多い）ので、全部タップして消すまでお邪魔は続く（居眠りはボールをはじいて目を覚ます）。お邪魔が終われば声も止まる。アタリはお邪魔の間も通常通り来るが、お邪魔中は合わせも餌付けもできない：「竿を持つ」ボタンと「餌付け」はお邪魔中は使えなくなり、水面・スペースキーでの合わせも効かない（竿を持っている最中にお邪魔が始まると、握りは解けて合わせにはならない）。本アタリは保持されず、通常と同じ持ち時間しかないので、ボール消しに手間取って合わせが遅れれば魚はエサを離して逃げる。釣りたければ、ボールを素早く全部消してお邪魔を終わらせ、本アタリが残っているうちに通常通り合わせること。ルアーが入ると魚が散るので、消えたら打ち直しになる（ただしアタリ進行中に消し終えた場合は、そのアタリの残り時間に任される）。
        8. 風景の音は設定でON/OFF。最初のタップで再生が始まる。OFFにすれば、YouTubeなどの他のアプリの音と一緒に楽しめる。音は合成ではなく、実際に野外で録音された環境音のループ：水のせせらぎがいつも流れ、早朝は鳥の合唱、昼は池の鳥の声（夏はセミの声も）、夕方は虫の声（夏・秋）、春・夏の朝と夕方にはカエルの合唱が、季節と時間帯に合わせて重なる。長さの違うループを重ねているので、同じ音の並びはめったに戻ってこない。録音出典（radio aporee／Internet Archive）：鳥の合唱＝Enquiring Ear（CC BY-NC 3.0）、池の鳥＝Carlos Santos（パブリックドメイン）、カエル＝リトアニア・ティルムーナイ（パブリックドメイン）、虫＝クロアチア・スプリト（パブリックドメイン）、セミと鳥＝Wu Tsan-Cheng（CC BY 3.0）、水音＝Vassar College（パブリックドメイン）。秋の池の写真：池と紅葉と茶室＝Yoshikazu TAKADA（CC BY 2.0）、ポートランド日本庭園＝King of Hearts（CC BY-SA 4.0）、モスクワ日本庭園＝Andrey Korzun（CC BY-SA 4.0）。波紋の写真＝Roger McLassus（CC BY-SA 3.0）。
        9. 距離は5メートル。ポイントは竿先から5.0m先。やり取りでは魚との距離がこの5.0mから近づいたり離れたりする。
        10. エサで釣れる魚がガラッと変わる。練り餌はフナ類・タナゴ中心ながら、鯉やブルーギルも来て、たまにカメまで釣れる。コーンは大型コイ・ソウギョ中心で小魚が激減、パンは表層の魚とニシキゴイ中心、ミミズはナマズ・ウナギなど肉食魚中心（カメもミミズが好きだ）。全31種のうち、今のエサではほぼ来ない魚も多いので、狙いを変えたらエサを変える。
        11. 正体は釣り上げるまで分からない。掛かった瞬間に魚種や大きさは表示されない。水中の姿も影だけで、タモに入れて初めて名前とサイズが分かる。
        12. リリースは優しく。鯉は濡れた手で触り、長く空気にさらさず、その場でそっと返すのがマナー。拾ったゴミは持ち帰ろう。ランキングに入る魚を登録せずにリリースしようとすると確認が出る（確認のボタンはリリースボタンと違う位置にあり、表示から600msは押せないので、連打での誤操作を防げる）。

        風景は画面いっぱいに広がり、下のボタン・状況の言葉は風景の上に重なって表示される。上部の水温などの情報は最初は隠れていて、上の「情報」ボタンで出し入れできる。状況の言葉が増えたり減ったりしても、風景そのものは決して動かない。延べ竿と糸は最初は表示されず、水面にはウキだけが立つ。竿と糸を見たいときは、設定の「竿と糸の表示」から表示できる。背景は実写の写真で、開くたびに季節と時間帯に合った写真（全19枚）からランダムに変わり、色づけの趣（24通り）も重なる。切り取り方や左右の向きまで変わるので、まるで違う場所で釣っているような気分になる（夏に紅葉、冬に桜のような季節外れの風景は出ない）。鳥居や橋、枝といった描き足しのイラストは一切ない。開いた時のデフォルトのステージは、端末のカレンダーと時計に合わせる：日付から季節（3〜5月は春、6〜8月は夏、9〜11月は秋、12〜2月は冬）を、現在時刻から時間帯（5〜10時台は早朝、11〜15時台は昼、16時〜翌4時台は夕方）を選び、その季節・時間帯の風景で始まる。設定で季節や時間帯を変えればその場で切り替わり、開き直すたびにまたカレンダーと時計のステージに戻る。設定で季節を変えると、その季節に合った風景に切り替わる。季節の情景も移ろう（春は桜、夏は日差し、秋は枯れ葉、冬は時々の雪）。待ちの間は、季節・時間帯・天気に寄せた独り言や「気のせいか？」といったウキの小さな揺れのコメント、ネットから取得した、地元のニュースの見出しと要約（出典・日付つき）への勝手な見解が流れ（取得できないときは、架空の世間話になる）、やり取り中は張りの状態に応じた掛け声が出る。ときどき魚が水面で跳ねることもある。ウキと水面だけを眺めて、静かに没頭してほしい。季節のいたずらもある：夏は蚊が飛んできて耳元を飛び回った末に刺される（かゆいが、釣りは止まらない）。秋はトンボ（アキアカネ）がウキのトップに止まって、その重みでウキが少し沈んでアタリが見えにくくなり、冬はウキに雪が少しずつ積もって、いつもより深く沈む（打ち直すか魚が引くと雪は落ちる）。

        サイズの目安：40〜60cmが中心、50cm超が最初の感動ライン、80cm超は記録級。水温15℃以上で活性が上がり、20〜25℃が最適。早朝と夕方が狙い目で、秋は越冬前の荒食いで一年で最も釣りやすい。

        延べ竿日和 ― ウキひとつ、水面ひとつ。

        1. Cast and wait. Choose where to cast by tapping the water: the rig flies to the very spot you tapped and the float stands there. Taps off the water — sky, hills, shore grass — do nothing (the “Cast the Rig” button casts to the current spot, the middle of the water the first time). A bite always comes within 10 seconds to 1.5 minutes. Casting again and again to the same spot draws fish in, and bites come sooner. When carp approach you may see small bubbles or an unnatural tremble of the float. When the wind crosses, ripples and a bright path run over the water and the float tilts downwind — the stronger the wind, the clearer the tilt.
        2. Read the bait from the float. With bait on, its weight sinks the float so the green band sits right at the surface. As the bait melts away the float rises, and when the red band shows above the water the bait is gone and no bite will come. Dough lasts about 1.5 minutes, bread about 45 seconds, earthworm about 2.5 minutes, and corn about 4 minutes. Press “Re-bait” to reel in and fit fresh bait — but the rig is not cast again yet. Once the new bait is on, the rig stays in hand until you tap the water, and it is cast to the very spot you tapped. Changing the bait in Settings works the same way: retrieve, hide the float once, fit the new bait, then tap the water to cast.
        3. Do not strike on the preliminary bite. Carp suck the bait in with the mud and spit it out first, so the float dips a few centimetres — that is the preliminary bite. Striking then usually misses, though a deeply sucked bait sometimes hooks anyway (more likely late in the dip). The proper way is to wait. But sometimes — roughly one time in three — there is no preliminary bite at all, and the real bite comes out of nowhere. Daydream, and you will miss the sink-away.
        4. Strike on the real bite — press, hold, then release. The button that appears after you cast is the “Hold the Rod” button. Press and hold it in advance to hold the rod, and release your finger at the instant of the real bite — the strike lands at the moment you let go. Pressing alone does not strike (while you hold, you are only holding the rod). When the float vanishes completely under the water, or rises up as the fish lifts the bait, let go. If you start holding late, during the bite itself, press and release at once to strike on the spot. While a sunken float stays fully under, the fish still holds the bait, so a slightly late release can still hook it. During the preliminary bite, keep holding and wait for the real bite. If a strong bite (the real bite) comes while you are not holding the rod and you still do not grab it, the rod will be yanked away into the pond and the whole rig is lost — you start over from the re-bait, so keep hold of the rod while you wait (during an interruption you cannot strike anyway, so the rod is safe then).
        5. Play the fish with the rod raised. There is no reel, so the line cannot run out. Hold “Raise the rod” to keep tension inside the green zone: while you hold the green, the fish comes much closer. Too much tension snaps the leader; too little and the hook comes out. Each fish is different: small fish forgive a wide range, while carp — especially over 50cm and 80cm — allow only a narrow range that moves fast. The ideal range is shown the moment the fish is hooked. When the tired fish is close, net it head-first.
        6. Buttons and water taps. Cast by tapping the water where you want the rig to land — after a re-bait, too, nothing is cast until you tap the water. Striking works by holding the rod: press and hold the “Hold the Rod” button, then release at the instant of the real bite to strike (the water works the same way — touching it holds the rod, letting go strikes; taps off the water do nothing. The space key also holds while pressed and strikes on release). During a fight, hold the raise button to add tension and release to ease off. While interrupted, the “Hold the Rod” button and “Re-bait” cannot be used (water and space-key strikes are off too) — clear all the balls first, then strike.
        7. Interruptions (always on). At exactly the same rate as bites — once every 10 seconds to 1.5 minutes after you start fishing, an old man or woman starts a long chat, a local kid throws a lure in, a stray cat, a camera-happy tourist, a senior angler, a video streamer or a pushy salesman shows up — or you doze off. Tap all 10 to 30 falling balls away to end the interruption (the more troublesome the visitor, the more balls). Each character's line is also spoken aloud, in a recorded voice fitted to the character — the old man low and slow, the old woman a little higher, the kid high and fast, the doze slow and sleepy, and so on. The voice stops when the interruption ends. Bites still arrive while you are interrupted, but you cannot strike or re-bait until the interruption ends: the “Hold the Rod” button and “Re-bait” are disabled, and water and space-key strikes do not work either (if an interruption starts while you are holding the rod, the grip is released without striking). A real bite is not held for you — it lasts only the usual time, so spend too long on the balls and the fish lets go and is gone. To catch, clear every ball quickly to end the interruption, then strike as usual while the real bite still lasts. A lure scatters the fish, so you will need to cast again (unless you clear it while a bite is in progress, in which case that bite simply keeps its remaining time).
        8. The sounds of the shore can be turned on and off in Settings. Sound starts with your first tap (turn it off to enjoy other apps' sound, such as YouTube, alongside the game), and it is not synthesized: it is real field recordings, looped and layered. Trickling water runs underneath everything; mornings bring a birdsong chorus, daytime the birds of a pond (with cicadas in summer), evenings the insects of summer and autumn, and spring and summer mornings and evenings a chorus of frogs — each following the season and time of day. The loops have different lengths, so the same sequence rarely comes round again.         9. The spot is 5 metres out. During a fight, the distance changes from that 5.0m as the fish runs and tires.
        10. Bait changes everything. Dough draws crucians and bitterlings — but carp and bluegill come too, and now and then even a turtle. Corn draws big carp and grass carp, bread draws surface fish and koi, and earthworm draws catfish, eels and other predators (turtles love earthworms too). Many of the 31 species rarely come to the wrong bait.
        11. You learn what it is only in the net. Species and size stay hidden while you fight; underwater you see only a shadow.
        12. Release gently. Handle carp with wet hands, keep them out of the air only briefly, and let them go right where you caught them. Take any litter home. If a fish would make the rankings, releasing it without registering asks for confirmation first (the confirmation buttons sit away from the release button, and cannot be tapped for 600ms after they appear, so rapid taps cannot confirm by accident).

        The scenery fills the screen, with the buttons and status words laid over it. The water temperature and other details at the top start hidden — use the “Info” button at the top to show or hide them. The rod and line also start hidden, so only the float stands on the water; you can show them any time under “Rod & Line” in Settings. The background is a real photograph, chosen at random on each open from 19 photos that fit the current season and time of day, layered with one of 24 colour moods — even the framing and the left-right orientation change, so it feels like fishing at a different place (you will never see autumn leaves in summer, or cherry blossoms in winter). Nothing is drawn in: no torii, no bridges, no branches — just the photograph. When you open the page, the default stage follows your device's calendar and clock: the season comes from the date (spring Mar–May, summer Jun–Aug, autumn Sep–Nov, winter Dec–Feb) and the time of day from the current hour (early morning 5–10, daytime 11–15, evening 16–4), and you start with scenery for that season and time. Changing them in Settings switches at once, and reopening the page returns to the calendar-and-clock stage. Changing the season in Settings switches to scenery that fits that season, and each season animates differently (cherry blossoms in spring, sunlight in summer, fallen leaves in autumn, occasional snow in winter). While you wait you will hear muttered thoughts, comments on tiny trembles of the float, and — fetched live from the web — local news headlines with summaries, source and date, plus the angler’s own unsolicited opinion (when the news cannot be fetched, the talk is openly fictional). Fish sometimes leap at the surface. Seasonal mischief too: in summer a mosquito may find you and bite (it itches, but the fishing goes on); in autumn a red dragonfly perches on the float tip and sinks it a little; in winter snow settles on the float and makes it sit deeper.

        Sizes: most carp run 40–60cm. Over 50cm is the first thrill; over 80cm is record class. Carp feed best at 15°C and above, ideally 20–25°C. Early morning and evening are prime times, and autumn — when carp feed hard before winter — is the easiest season of all.

        Nobezao Biyori — A Fine Day for Pole Fishing. One float, one sheet of water.
